/** Nhận diện mã quốc gia từ chuỗi người dùng gõ tay — khớp mã, khớp tên chính xác (bỏ dấu), rồi khớp gần đúng */
function findCountryCode(raw){
  const norm = s => (s||"").trim().toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g,'');
  const target = norm(raw);
  if(!target) return null;
  let hit = COUNTRY_LIST.find(c => c.code === target);
  if(hit) return hit.code;
  hit = COUNTRY_LIST.find(c => norm(c.name) === target);
  if(hit) return hit.code;
  hit = COUNTRY_LIST.find(c => norm(c.name).includes(target) || target.includes(norm(c.name)));
  return hit ? hit.code : null;
}

}

/** Áp dụng tọa độ của địa điểm đã chọn (chip hoặc gõ tay khớp) rồi tìm cơ sở gần nhất luôn */

/** Tự động xác định vị trí thật của người dùng bằng Geolocation API của trình duyệt.
 *  Không cần người dùng nhập bất kỳ thông tin nào — thành công thì tự gọi findNearest(). */
function detectUserLocation(){
  const statusEl = document.getElementById('geoStatus');
  if(!('geolocation' in navigator)){
    statusEl.textContent = '⚠ Thiết bị/trình duyệt không hỗ trợ định vị GPS. Hãy dùng mục "Nâng cao" bên dưới để mô phỏng vị trí.';
    return;
  }
  statusEl.textContent = '📍 Đang tự động xác định vị trí của bạn...';
  navigator.geolocation.getCurrentPosition(
    pos=>{
      document.getElementById('userLat').value = pos.coords.latitude.toFixed(4);
      document.getElementById('userLng').value = pos.coords.longitude.toFixed(4);
      statusEl.textContent = `✅ Đã tự động xác định vị trí (độ chính xác ~${Math.round(pos.coords.accuracy)}m). Đang tìm cơ sở y tế gần nhất...`;
      findNearest();
    },
    err=>{
      statusEl.textContent = `⚠ Không lấy được vị trí tự động (${err.message}). Hãy mở mục "Nâng cao" bên dưới để mô phỏng vị trí, hoặc cấp quyền định vị rồi nhấn "Xác định lại vị trí GPS".`;
    },
    { enableHighAccuracy:true, timeout:10000, maximumAge:60000 }
  );
}
function haversine(lat1,lng1,lat2,lng2){
  const R=6371; const toRad=d=>d*Math.PI/180;
  const dLat=toRad(lat2-lat1), dLng=toRad(lng2-lng1);
  const a=Math.sin(dLat/2)**2 + Math.cos(toRad(lat1))*Math.cos(toRad(lat2))*Math.sin(dLng/2)**2;
  return R*2*Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}
/** Tính góc phương vị (bearing, độ) từ điểm 1 tới điểm 2 */
function computeBearing(lat1,lng1,lat2,lng2){
  const toRad=d=>d*Math.PI/180, toDeg=r=>r*180/Math.PI;
  const dLng = toRad(lng2-lng1);
  const y = Math.sin(dLng)*Math.cos(toRad(lat2));
  const x = Math.cos(toRad(lat1))*Math.sin(toRad(lat2)) - Math.sin(toRad(lat1))*Math.cos(toRad(lat2))*Math.cos(dLng);
  return (toDeg(Math.atan2(y,x))+360)%360;
}
/** Quy đổi góc độ sang 8 hướng la bàn tiếng Việt */
function bearingToCompass(deg){
  const dirs=['Bắc','Đông Bắc','Đông','Đông Nam','Nam','Tây Nam','Tây','Tây Bắc'];
  return dirs[Math.round(deg/45)%8];
}

/* ---- Thời tiết theo vị trí hiện tại (Open-Meteo — miễn phí, không cần API key) ---- */
let lastWeatherLat = null, lastWeatherLng = null;
async function fetchWeather(lat, lng){
  lastWeatherLat = lat; lastWeatherLng = lng;
  const label = document.getElementById('weatherLabel');
  if(!label) return;
  try{
    const res = await fetch(`https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lng}&current_weather=true`);
    if(!res.ok) throw new Error('Không lấy được dữ liệu thời tiết');
    const data = await res.json();
    const c = data.current_weather.temperature;
    const f = c * 9/5 + 32;
    label.textContent = `${c.toFixed(1)}°C / ${f.toFixed(1)}°F`;
  } catch(err){
    label.textContent = 'Chưa có dữ liệu thời tiết (cần mạng)';
  }
}
/** Làm mới lại nhiệt độ mỗi 10 phút cho cùng 1 vị trí, để bắt kịp thời tiết thay đổi theo thời gian */
function refreshWeather(){ if(lastWeatherLat!==null) fetchWeather(lastWeatherLat, lastWeatherLng); }

let lastDirectionSteps = [];
function findNearest(){
  const lat=parseFloat(document.getElementById('userLat').value), lng=parseFloat(document.getElementById('userLng').value);
  fetchWeather(lat, lng);
  const ranked = facilities.map(f=>({...f, dist:haversine(lat,lng,f.lat,f.lng)})).sort((a,b)=>a.dist-b.dist);
  const top3 = ranked.slice(0,3);
  const list = document.getElementById('facilityList'); list.innerHTML='';
  top3.forEach(f=>{
    const d = document.createElement('div'); d.className='facility-item';
    d.innerHTML = `<div><div>${f.name}</div><div class="type">${f.type} · ${f.country}</div></div><div class="dist">${f.dist.toFixed(1)} km</div>`;
    list.appendChild(d);
  });
  const dest = top3[0];
  // Tạo điểm trung gian (waypoint) lệch nhẹ để mô phỏng một tuyến đường thực tế thay vì đường thẳng tuyệt đối
  const mid = { lat:(lat+dest.lat)/2 + (dest.lng-lng)*0.06, lng:(lng+dest.lng)/2 - (dest.lat-lat)*0.06 };
  drawMap(lat,lng,top3,{mid,dest});

  const leg1Dist = haversine(lat,lng,mid.lat,mid.lng), leg1Bear = computeBearing(lat,lng,mid.lat,mid.lng);
  const leg2Dist = haversine(mid.lat,mid.lng,dest.lat,dest.lng), leg2Bear = computeBearing(mid.lat,mid.lng,dest.lat,dest.lng);
  lastDirectionSteps = [
    `Cơ sở y tế gần nhất là ${dest.name}, cách bạn khoảng ${dest.dist.toFixed(1)} kilômét.`,
    `Bước 1: di chuyển khoảng ${leg1Dist.toFixed(1)} kilômét theo hướng ${bearingToCompass(leg1Bear)}.`,
    `Bước 2: tiếp tục khoảng ${leg2Dist.toFixed(1)} kilômét theo hướng ${bearingToCompass(leg2Bear)} để đến ${dest.name}.`,
    `Bạn đã đến nơi. Hãy tìm nhân viên y tế để được hỗ trợ ngay.`
  ];
  document.getElementById('directionsCard').style.display='block';
  const stepsDiv = document.getElementById('directionSteps'); stepsDiv.innerHTML='';
  lastDirectionSteps.forEach((s,i)=>{
    const seg = document.createElement('div'); seg.className='route-seg';
    seg.innerHTML = `<span class="seg-badge ok">${i===0?'ĐÍCH':'B.'+i}</span><span>${s}</span>`;
    stepsDiv.appendChild(seg);
  });
  queueSync();
}

/* ---- Zoom (phóng to/thu nhỏ) & Pan (kéo/lướt) cho bản đồ vector ---- */
/* Khai báo TRƯỚC drawMap() vì drawMap() sẽ gọi resetMapView() mỗi lần vẽ lại dữ liệu mới */
const MAP_W = 600, MAP_H = 220;
const MAP_ZOOM_MIN_W = 120; // giới hạn phóng to tối đa (~5x)
let mapViewBox = { x:0, y:0, w:MAP_W, h:MAP_H };
function applyMapViewBox(){
  const svgEl = document.getElementById('mapSvg');
  if(svgEl) svgEl.setAttribute('viewBox', `${mapViewBox.x} ${mapViewBox.y} ${mapViewBox.w} ${mapViewBox.h}`);
}
function clampMapViewBox(){
  mapViewBox.w = Math.min(MAP_W, Math.max(MAP_ZOOM_MIN_W, mapViewBox.w));
  mapViewBox.h = mapViewBox.w * (MAP_H / MAP_W);
  mapViewBox.x = Math.min(Math.max(mapViewBox.x, 0), Math.max(0, MAP_W - mapViewBox.w));
  mapViewBox.y = Math.min(Math.max(mapViewBox.y, 0), Math.max(0, MAP_H - mapViewBox.h));
}
/** factor < 1 = phóng to (zoom in), factor > 1 = thu nhỏ (zoom out) — giữ nguyên tâm khung nhìn hiện tại */
function zoomMap(factor){
  const cx = mapViewBox.x + mapViewBox.w/2, cy = mapViewBox.y + mapViewBox.h/2;
  mapViewBox.w *= factor;
  mapViewBox.w = Math.min(MAP_W, Math.max(MAP_ZOOM_MIN_W, mapViewBox.w));
  mapViewBox.h = mapViewBox.w * (MAP_H / MAP_W);
  mapViewBox.x = cx - mapViewBox.w/2;
  mapViewBox.y = cy - mapViewBox.h/2;
  clampMapViewBox();
  applyMapViewBox();
}
function resetMapView(){ mapViewBox = { x:0, y:0, w:MAP_W, h:MAP_H }; applyMapViewBox(); }

function drawMap(userLat,userLng,highlighted,routeInfo){
  const svg = document.getElementById('mapSvg');
  const allLats = facilities.map(f=>f.lat).concat(userLat), allLngs = facilities.map(f=>f.lng).concat(userLng);
  if(routeInfo){ allLats.push(routeInfo.mid.lat, routeInfo.dest.lat); allLngs.push(routeInfo.mid.lng, routeInfo.dest.lng); }
  const minLat=Math.min(...allLats), maxLat=Math.max(...allLats), minLng=Math.min(...allLngs), maxLng=Math.max(...allLngs);
  const pad=20, W=600, H=220;
  const x = lng => pad + (lng-minLng)/(maxLng-minLng||1) * (W-2*pad);
  const y = lat => H-pad - (lat-minLat)/(maxLat-minLat||1) * (H-2*pad);
  let s='';
  facilities.forEach(f=>{
    const isTop = highlighted && highlighted.some(h=>h.name===f.name);
    s += `<circle cx="${x(f.lng)}" cy="${y(f.lat)}" r="${isTop?6:4}" class="dot-fac" opacity="${isTop?1:0.5}"/>`;
    if(isTop) s += `<text x="${x(f.lng)+8}" y="${y(f.lat)+4}" fill="#8992a0" font-size="10">${f.name}</text>`;
  });
  if(routeInfo){
    const ux=x(userLng), uy=y(userLat), mx=x(routeInfo.mid.lng), my=y(routeInfo.mid.lat), dx=x(routeInfo.dest.lng), dy=y(routeInfo.dest.lat);
    s += `<polyline points="${ux},${uy} ${mx},${my} ${dx},${dy}" fill="none" stroke="#ff5a2b" stroke-width="2.2" stroke-dasharray="6 4"/>`;
    s += `<circle cx="${mx}" cy="${my}" r="3" fill="#ff5a2b"/>`;
  }
  s += `<circle cx="${x(userLng)}" cy="${y(userLat)}" r="7" class="dot-user"/><text x="${x(userLng)+9}" y="${y(userLat)-8}" fill="#ff5a2b" font-size="11" font-weight="bold">BẠN</text>`;
  svg.innerHTML = s;
  resetMapView(); // mỗi lần vẽ lại dữ liệu mới thì đưa khung nhìn về mặc định
}

(function initMapPanZoom(){
  const svg = document.getElementById('mapSvg');
  const pointers = new Map(); // pointerId -> {x, y}
  let dragStart = null;      // {x, y, vb} — dùng khi chỉ có 1 điểm chạm/chuột (kéo/lướt)
  let pinchStartDist = null; // khoảng cách giữa 2 ngón lúc bắt đầu chụm (dùng khi có 2 điểm chạm)

  function dist(a, b){ return Math.hypot(a.x - b.x, a.y - b.y); }
  function currentPoints(){ return [...pointers.values()]; }

  function onPointerDown(e){
    svg.setPointerCapture(e.pointerId);
    pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
    if(pointers.size === 1){
      dragStart = { x: e.clientX, y: e.clientY, vb: { ...mapViewBox } };
      pinchStartDist = null;
    } else if(pointers.size === 2){
      const [a, b] = currentPoints();
      pinchStartDist = dist(a, b);
      dragStart = null;
    }
  }
  function onPointerMove(e){
    if(!pointers.has(e.pointerId)) return;
    pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });

    if(pointers.size === 1 && dragStart){
      // Kéo/lướt bằng 1 ngón tay hoặc chuột — trái/phải/lên/xuống
      const rect = svg.getBoundingClientRect();
      const scaleX = mapViewBox.w / rect.width, scaleY = mapViewBox.h / rect.height;
      mapViewBox.x = dragStart.vb.x - (e.clientX - dragStart.x) * scaleX;
      mapViewBox.y = dragStart.vb.y - (e.clientY - dragStart.y) * scaleY;
      clampMapViewBox();
      applyMapViewBox();
    } else if(pointers.size === 2 && pinchStartDist !== null){
      // Chụm/xòe 2 ngón tay để zoom out/in trên điện thoại, máy tính bảng
      const [a, b] = currentPoints();
      const newDist = dist(a, b);
      if(newDist > 0 && pinchStartDist > 0){
        zoomMap(pinchStartDist / newDist); // 2 ngón tách xa nhau -> newDist lớn hơn -> factor<1 -> phóng to
        pinchStartDist = newDist; // cập nhật liên tục để zoom mượt theo từng khung hình
      }
    }
    e.preventDefault();
  }
  function onPointerUp(e){
    pointers.delete(e.pointerId);
    if(pointers.size === 1){
      const [p] = currentPoints();
      dragStart = { x: p.x, y: p.y, vb: { ...mapViewBox } };
      pinchStartDist = null;
    } else {
      dragStart = null;
      pinchStartDist = null;
    }
  }

  svg.addEventListener('pointerdown', onPointerDown);
  svg.addEventListener('pointermove', onPointerMove, { passive:false });
  svg.addEventListener('pointerup', onPointerUp);
  svg.addEventListener('pointercancel', onPointerUp);

  // Lăn chuột (bánh xe) để zoom trên desktop
  svg.addEventListener('wheel', e => { e.preventDefault(); zoomMap(e.deltaY > 0 ? 1.15 : 0.87); }, { passive:false });
})();

/* ============ MOD 4: SAFE ROUTE PREDICTOR (Dijkstra, đồ thị động theo từng quốc gia) ============ */
let routeSelectedCountry = null;
let graph = {};
let edgeSignal = {};
/** Khoảng cách "đường thực tế" ước tính = đường chim bay x hệ số đường đi, làm tròn 1 số thập phân */
function haversineRoad(lat1,lng1,lat2,lng2){ return +(haversine(lat1,lng1,lat2,lng2) * 1.15).toFixed(1); }
/** Sắp xếp 8 địa điểm thành 1 chuỗi lộ trình theo nguyên tắc "lân cận gần nhất" (nearest-neighbor),
 *  để khoảng cách giữa các chặng liên tiếp thực tế thay vì nhảy lung tung khắp quốc gia */
function buildNearestNeighborChain(nodes){
  const remaining = nodes.slice();
  const order = [remaining.shift()];
  while(remaining.length){
    const last = order[order.length-1];
    let bestIdx = 0, bestDist = Infinity;
    remaining.forEach((n,i)=>{ const d = haversine(last.lat,last.lng,n.lat,n.lng); if(d<bestDist){ bestDist=d; bestIdx=i; } });
    order.push(remaining.splice(bestIdx,1)[0]);
  }
  return order;
}
function getSignal(a,b){ return edgeSignal[a+"-"+b] || edgeSignal[b+"-"+a] || "ok"; }
function dijkstra(graph,start,end){
  const dist={}, prev={}, visited=new Set();
  Object.keys(graph).forEach(n=>dist[n]=Infinity); dist[start]=0;
  while(true){
    let u=null,best=Infinity;
    for(const n in dist){ if(!visited.has(n) && dist[n]<best){ best=dist[n]; u=n; } }
    if(u===null||u===end) break;
    visited.add(u);
    for(const v in graph[u]){ const alt=dist[u]+graph[u][v]; if(alt<dist[v]){ dist[v]=alt; prev[v]=u; } }
  }
  const path=[]; let cur=end; while(cur){ path.unshift(cur); cur=prev[cur]; }
  return { path, total: dist[end] };
}
/** Khi người dùng gõ/chọn quốc gia — dùng chung dữ liệu địa điểm với MOD.02 (countryLocations) */
function onRouteCountryChange(){
  const code = findCountryCode(document.getElementById('routeCountryInput').value);
  buildRouteGraphForCountry(code);
}
/** Dựng đồ thị 8 địa điểm (4 đô thị + 4 hẻo lánh) của quốc gia, xen kẽ thành 1 chuỗi lộ trình để Dijkstra có nhiều chặng để tính */
function buildRouteGraphForCountry(code){
  const card = document.getElementById('routePickerCard');
  document.getElementById('routeResult').style.display = 'none';
  if(!code || !countryLocations[code]){ card.style.display='none'; routeSelectedCountry=null; return; }
  routeSelectedCountry = code;
  const loc = countryLocations[code];
  const countryName = (COUNTRY_LIST.find(c=>c.code===code)||{}).name || code;
  document.getElementById('routeCountryName').textContent = countryName;

  const nodes = [];
  loc.urban.forEach(u => nodes.push({ ...u, type:'urban' }));
  loc.remote.forEach(r => nodes.push({ ...r, type:'remote' }));
  const order = buildNearestNeighborChain(nodes);
  graph = {}; edgeSignal = {};
  order.forEach(n=>{ graph[n.name] = {}; });
  for(let i=0;i<order.length-1;i++){
    const a = order[i], b = order[i+1];
    const km = haversineRoad(a.lat,a.lng,b.lat,b.lng);
    graph[a.name][b.name] = km; graph[b.name][a.name] = km;
    edgeSignal[a.name+'-'+b.name] = (a.type==='urban' && b.type==='urban') ? 'ok' : 'dead';
  }

  const dl = document.getElementById('routeNodesList');
  dl.innerHTML = order.map(n=>`<option value="${n.name}">`).join('');
  document.getElementById('routeFrom').value = order[0].name;
  document.getElementById('routeTo').value = order[order.length-1].name;
  card.style.display = 'block';
}
/** Khớp tên địa điểm người dùng gõ với node có sẵn trong đồ thị của quốc gia đang chọn */
function findGraphNode(input){
  const norm = s=>s.trim().toLowerCase();
  const target = norm(input);
  if(!target) return null;
  const keys = Object.keys(graph);
  let match = keys.find(k=>norm(k)===target);
  if(!match) match = keys.find(k=>norm(k).includes(target) || target.includes(norm(k)));
  return match || null;
}
function computeRoute(){
  if(!routeSelectedCountry) return;
  const fromInput = document.getElementById('routeFrom').value, toInput = document.getElementById('routeTo').value;
  const from = findGraphNode(fromInput), to = findGraphNode(toInput);
  document.getElementById('routeResult').style.display='block';
  const segsDiv = document.getElementById('routeSegs'); segsDiv.innerHTML='';
  const warnDiv = document.getElementById('routeWarn'); warnDiv.innerHTML='';
  if(!from || !to){
    warnDiv.innerHTML = `<div class="warn-banner">⚠ Không tìm thấy "${!from?fromInput:toInput}" trong danh sách địa điểm của quốc gia này. Hãy chọn đúng tên từ gợi ý.</div>`;
    return;
  }
  if(from===to){
    warnDiv.innerHTML = `<div class="warn-banner">⚠ Điểm đi và điểm đến đang trùng nhau. Hãy chọn 2 địa điểm khác nhau.</div>`;
    return;
  }
  const { path, total } = dijkstra(graph, from, to);
  let deadKm = 0;
  for(let i=0;i<path.length-1;i++){
    const a=path[i], b=path[i+1], km=graph[a][b], sig=getSignal(a,b);
    if(sig==='dead') deadKm += km;
    const seg = document.createElement('div'); seg.className='route-seg'; seg.style.flexWrap='wrap';
    seg.innerHTML = `<span class="seg-badge ${sig==='ok'?'ok':'dead'}">${sig==='ok'?'CÓ SÓNG':'MẤT SÓNG'}</span><span>${a} → ${b}</span><span style="color:var(--text-muted);margin-left:auto;">${km.toFixed(1)} km</span>`;
    if(sig==='dead'){
      const btn = document.createElement('button');
      btn.className = 'ghost'; btn.style.marginLeft = '8px'; btn.style.fontSize = '12px';
      btn.textContent = '⬇ Tải dữ liệu offline (cơ sở y tế + bản đồ)';
      btn.onclick = () => downloadOfflinePack(btn);
      seg.appendChild(btn);
    }
    segsDiv.appendChild(seg);
  }
  warnDiv.innerHTML = `<p style="color:var(--text-muted);font-size:13px;margin-top:10px;">Tổng quãng đường: ${total.toFixed(1)} km</p>`;
  if(deadKm>0) warnDiv.innerHTML += `<div class="warn-banner">⚠ Lộ trình có ${deadKm.toFixed(1)} km mất sóng. Nên tải trước dữ liệu offline cho các đoạn được đánh dấu bên trên trước khi khởi hành.</div>`;
}
/** Mô phỏng tải gói dữ liệu offline (cơ sở y tế + bản đồ vector) cho 1 chặng mất sóng */

  btn.disabled = true;
  btn.textContent = '✓ Đã tải xong (~5MB)';
  queueSync();
}

function renderSearchResults(){
  const q = document.getElementById('faSearch').value.trim().toLowerCase();
  const resultsWrap = document.getElementById('searchResults'); resultsWrap.innerHTML='';
  if(!q) return;
  const matches = Object.keys(firstAidDB).filter(id=>{
    const entry = firstAidDB[id];
    return entry.name.toLowerCase().includes(q) || entry.tags.some(t=>t.toLowerCase().includes(q));
  });
  if(matches.length===0){
    resultsWrap.innerHTML = '<span class="desc" style="margin:0;">Không tìm thấy — thử từ khóa khác (vd: "gãy", "bỏng", "dị ứng"...).</span>';
    return;
  }
  matches.forEach(id=>{
    const b = document.createElement('button'); b.className='chip'; b.textContent = `${firstAidDB[id].icon||''} ${firstAidDB[id].name}`;
    b.onclick = ()=> selectFirstAid(id);
    resultsWrap.appendChild(b);
  });
}

function selectFirstAid(id){
  const entry = firstAidDB[id];
  document.getElementById('faEmpty').style.display = 'none';
  document.getElementById('fa-cpr').style.display = 'none';
  document.getElementById('fa-generic').style.display = 'none';

}

// --- Bộ hiển thị bước dùng chung (generic step viewer), hỗ trợ đa ngôn ngữ ---
const genericState = { id:null, step:0 };
const faLangNames = { vi:"Tiếng Việt", en:"English", es:"Español", fr:"Français", zh:"中文", ja:"日本語", ko:"한국어", th:"ไทย", lo:"ລາວ", km:"ខ្មែរ", tl:"Filipino", ms:"Bahasa Melayu", np:"नेपाली", is:"Íslenska", ar:"العربية", sw:"Kiswahili" };
/** Lấy tên + các bước theo ngôn ngữ đang chọn. Nếu chưa có gói dịch, dùng English làm dự phòng
 *  và báo rõ cho người dùng biết (thay vì âm thầm hiển thị tiếng Việt gây hiểu nhầm). */
function getLocalizedEntry(entry, lang){
  if(lang==='vi') return { name:entry.name, steps:entry.steps, usedFallback:false };
  if(entry.translations && entry.translations[lang] && entry.translations[lang].steps){
    const t = entry.translations[lang];
    return { name:t.name, steps:t.steps, usedFallback:false };
  }
  // Dự phòng: dùng English nếu có, nếu không thì tiếng Việt gốc
  if(entry.translations && entry.translations.en && entry.translations.en.steps){
    return { name:entry.translations.en.name, steps:entry.translations.en.steps, usedFallback:true };
  }
  return { name:entry.name, steps:entry.steps, usedFallback:true };
}
function renderGenericStep(){
  if(!genericState.id) return;
  const entry = firstAidDB[genericState.id];
  const countryCode = findCountryCode(document.getElementById('faCountry').value);
  const lang = countryToLang[countryCode] || 'en';
  const loc = getLocalizedEntry(entry, lang);
  const idx = genericState.step;
  document.getElementById('faGenericIcon').textContent = entry.icon || '🩹';
  document.getElementById('faGenericTitle').textContent = loc.name.toUpperCase();
  document.getElementById('faGenericText').textContent = `${idx+1}/${loc.steps.length}: ${loc.steps[idx]}`;
  renderIllustration(genericState.id, idx);
  const dotsWrap = document.getElementById('faGenericDots'); dotsWrap.innerHTML = '';
  loc.steps.forEach((_,i)=>{ const d=document.createElement('div'); d.className='step-dot'+(i<=idx?' on':''); dotsWrap.appendChild(d); });
  document.getElementById('faVideoLink').href = videoSearchUrl(entry.translations && entry.translations.en ? entry.translations.en.name : entry.name);
  const noteEl = document.getElementById('faLangFallbackNote');
  if(!countryCode){
    noteEl.style.display = 'block';
    noteEl.textContent = '⚠ Không nhận diện được quốc gia này — đang hiển thị tạm bằng English. Hãy gõ đúng tên hoặc chọn từ danh sách gợi ý.';
  } else if(loc.usedFallback && lang!=='vi'){
    noteEl.style.display = 'block';
    noteEl.textContent = `🌐 Gói dịch chi tiết cho ${faLangNames[lang]||lang} đang được bổ sung — đang hiển thị tạm bằng English.`;
  } else {
    noteEl.style.display = 'none';
  }
}
function nextGenericStep(){
  const entry = firstAidDB[genericState.id];
  const countryCode = findCountryCode(document.getElementById('faCountry').value);
  const lang = countryToLang[countryCode] || 'en';
  const loc = getLocalizedEntry(entry, lang);
  if(genericState.step < loc.steps.length-1){ genericState.step++; renderGenericStep(); }
}
function prevGenericStep(){
  if(genericState.step > 0){ genericState.step--; renderGenericStep(); }
}

// --- Nhận diện giọng nói (Web Speech API) ---
function startVoiceSearch(){
  const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
  const statusEl = document.getElementById('voiceStatus');
  if(!SpeechRecognition){
    statusEl.textContent = 'Trình duyệt này không hỗ trợ nhận diện giọng nói. Hãy nhập từ khóa thủ công.';
    return;
  }
  const recognition = new SpeechRecognition();
  recognition.lang = 'vi-VN';
  recognition.interimResults = false;
  recognition.maxAlternatives = 1;
  statusEl.textContent = '🎙️ Đang nghe... hãy nói tên chấn thương.';
  recognition.start();
  recognition.onresult = (e)=>{
    const transcript = e.results[0][0].transcript;
    document.getElementById('faSearch').value = transcript;
    statusEl.textContent = `Đã nghe: "${transcript}"`;
    renderSearchResults();
  };
  recognition.onerror = ()=>{ statusEl.textContent = 'Không nhận diện được giọng nói, hãy thử lại hoặc nhập tay.'; };
  recognition.onend = ()=>{ if(statusEl.textContent.includes('Đang nghe')) statusEl.textContent=''; };
}

// --- CPR (đặc biệt: hoạt ảnh + đếm giờ + âm thanh) ---
let cprInterval=null, cprAudioCtx=null, cprTimeLeft=30, cprCount=0;
function beep(){
  if(!cprAudioCtx) return;
  const osc = cprAudioCtx.createOscillator(), gain = cprAudioCtx.createGain();
  osc.frequency.value = 880; osc.connect(gain); gain.connect(cprAudioCtx.destination);
  gain.gain.setValueAtTime(0.15, cprAudioCtx.currentTime);
  osc.start(); osc.stop(cprAudioCtx.currentTime + 0.08);
}
function startCPR(){
  if(cprInterval) return;
  cprAudioCtx = cprAudioCtx || new (window.AudioContext||window.webkitAudioContext)();
  cprTimeLeft = 30; cprCount = 0;
  document.getElementById('cprHands').style.animationPlayState = 'running';
  updateCprUI();
  cprInterval = setInterval(()=>{
    cprCount++; beep(); cprTimeLeft -= 0.545;
    if(cprTimeLeft <= 0){ stopCPR(true); }
    updateCprUI();
  }, 545);
}
function stopCPR(done){
  clearInterval(cprInterval); cprInterval = null;
  document.getElementById('cprHands').style.animationPlayState = 'paused';
  if(done) queueSync();
}
function updateCprUI(){
  document.getElementById('cprTimer').textContent = Math.max(0, Math.ceil(cprTimeLeft)) + 's';
  document.getElementById('cprCount').textContent = cprCount;
}
